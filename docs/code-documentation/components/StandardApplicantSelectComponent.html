import {
  ChangeDetectionStrategy,
  Component,
  EnvironmentInjector,
  OnChanges,
  OnInit,
  SimpleChanges,
  inject,
  input,
  output,
  signal,
} from '@angular/core';
import {
  FormControl,
  FormGroup,
  ReactiveFormsModule,
  Validators,
} from '@angular/forms';

import { STANDARD_APPLICANT_SEARCH_ERROR_MESSAGES } from './util/error-messages';
import {
  mapSaToRow,
  standardAppColumns,
} from './util/standard-applicant-select-row-helpers';
import {
  StandardApplicantSelectPagingState,
  initialStandardApplicantSelectPagingState,
} from './util/standard-applicant-select.state';

import { ErrorItem } from '@components/error-summary/error-summary.component';
import { PaginationComponent } from '@components/pagination/pagination.component';
import {
  SortableTableComponent,
  TableColumn,
} from '@components/sortable-table/sortable-table.component';
import { TextInputComponent } from '@components/text-input/text-input.component';
import {
  GetStandardApplicantsRequestParams,
  StandardApplicantsApi,
} from '@openapi';
import { ErrorMessageMap, buildFormErrorSummary } from '@util/error-summary';
import { createSignalState, setupLoadEffect } from '@util/signal-state-helpers';
import { toStandardApplicantSortKey } from '@util/standard-applicant-sort-map';
import { StandardApplicantRow } from '@util/types/applications-list-entry/types';

export type SelectedStandardApplicantSummary = {
  standardApplicantId?: number;
  code: string;
  name: string;
};

type StandardApplicantFilters = Pick<
  GetStandardApplicantsRequestParams,
  'code' | 'name'
>;

@Component({
  selector: 'app-standard-applicant-select',
  standalone: true,
  imports: [
    ReactiveFormsModule,
    TextInputComponent,
    SortableTableComponent,
    PaginationComponent,
  ],
  changeDetection: ChangeDetectionStrategy.Eager,
  templateUrl: './standard-applicant-select.component.html',
})
export class StandardApplicantSelectComponent implements OnInit, OnChanges {
  private readonly saApi = inject(StandardApplicantsApi);
  private readonly envInjector = inject(EnvironmentInjector);

  selectedCode = input<string | null>(null);
  readonly selectedId = input<number | null>(null);
  readonly initialCodeFilter = input<string | null>(null);
  readonly selectedCodeChange = output<string | null>();
  readonly selectedApplicantSummaryChange =
    output<SelectedStandardApplicantSummary | null>();
  readonly searchErrorsChange = output<ErrorItem[]>();

  rows: StandardApplicantRow[] = [];

  readonly columns: TableColumn[] = standardAppColumns;

  // Paging
  private readonly saSignalState =
    createSignalState<StandardApplicantSelectPagingState>(
      initialStandardApplicantSelectPagingState,
    );
  private readonly saState = this.saSignalState.state;
  readonly vm = this.saSignalState.vm;

  private readonly loadRequest =
    signal<GetStandardApplicantsRequestParams | null>(null);
  readonly submitted = signal(false);
  private readonly errorMap: ErrorMessageMap =
    STANDARD_APPLICANT_SEARCH_ERROR_MESSAGES;
  private appliedFilters: StandardApplicantFilters = {};
  private initialLookupPending = false;
  private selectedRowCode: string | null = null;

  form = new FormGroup({
    code: new FormControl<string>('', {
      nonNullable: true,
      validators: [Validators.maxLength(10)],
    }),
    name: new FormControl<string>('', {
      nonNullable: true,
      validators: [Validators.maxLength(100)],
    }),
  });

  // Selection for the table
  selectedIds: Set<string> = new Set<string>();

  ngOnInit(): void {
    const code = this.initialCodeFilter()?.trim();
    if (code) {
      this.form.controls.code.setValue(code);
      this.appliedFilters = { code };
    }
    this.initialLookupPending = !!code;
    this.syncSelectedIds();
    this.setupEffects();
    this.saSignalState.patch({ hasSearched: true });
    this.loadPage(0);
  }

  ngOnChanges(changes: SimpleChanges): void {
    if ('selectedId' in changes || 'selectedCode' in changes) {
      // Preserve an explicit click when the parent echoes its code back.
      if (
        !this.selectedCode() ||
        ('selectedId' in changes && this.selectedId() !== null) ||
        this.selectedRowCode !== this.selectedCode()
      ) {
        this.syncSelectedIds();
      }
    }
  }

  onSelectedIdsChange(ids: Set<string>): void {
    const first = ids.values().next().value;
    const selectedRow =
      first === undefined
        ? null
        : (this.rows.find(
            (row) =>
              typeof row.standardApplicantId === 'number' &&
              String(row.standardApplicantId) === first,
          ) ?? null);
    // Missing IDs must never fall back to code identity or clear saved data.
    if (first !== undefined && !selectedRow) {
      return;
    }
    this.selectedIds = ids;
    const code = selectedRow?.code ?? null;
    this.selectedRowCode = code;

    if (code !== this.selectedCode()) {
      this.selectedCodeChange.emit(code);
    }

    this.selectedApplicantSummaryChange.emit(
      selectedRow
        ? {
            code: selectedRow.code ?? '',
            standardApplicantId: selectedRow.standardApplicantId,
            name: selectedRow.name ?? '',
          }
        : null,
    );
  }

  onPageChange(page: number): void {
    this.initialLookupPending = false;
    if (!this.saState().hasSearched) {
      return;
    }
    this.loadPage(page);
  }

  onSortChange(sort: { key: string; direction: 'desc' | 'asc' }): void {
    this.initialLookupPending = false;
    if (!this.saState().hasSearched) {
      return;
    }
    this.saSignalState.patch({ sortField: sort, pageIndex: 0 });
    this.loadPage(0);
  }

  onSubmit(event: SubmitEvent): void {
    this.initialLookupPending = false;
    event.preventDefault();
    this.submitted.set(true);
    this.form.markAllAsTouched();
    this.form.updateValueAndValidity({ emitEvent: false });

    const validationErrors = this.buildErrorSummary();
    this.saSignalState.patch({ searchErrors: validationErrors });
    this.searchErrorsChange.emit(validationErrors);

    if (validationErrors.length) {
      return;
    }

    this.appliedFilters = this.getTrimmedFilters();
    this.saSignalState.patch({ hasSearched: true });
    this.loadPage(0, this.appliedFilters);
  }

  fieldError(id: string): ErrorItem | undefined {
    return this.vm().searchErrors.find((e) => e.id === id);
  }

  clearSearch(): void {
    this.initialLookupPending = false;
    this.loadRequest.set(null);
    this.form.reset();
    this.appliedFilters = {};
    this.rows = [];
    this.saSignalState.patch({
      hasSearched: true,
      pageIndex: 0,
      totalPages: 0,
      loading: false,
      searchErrors: [],
    });
    this.searchErrorsChange.emit([]);
    this.loadPage(0, this.appliedFilters);
  }

  private setupEffects(): void {
    setupLoadEffect(
      {
        request: this.loadRequest,
        load: (params) =>
          this.saApi.getStandardApplicants(params, 'body', false, {
            transferCache: true,
          }),
        onSuccess: (page) => {
          const content = page.content ?? [];
          this.rows = content.map((sa) => mapSaToRow(sa));

          this.saSignalState.patch({
            totalPages: page.totalPages,
            loading: false,
            searchErrors: [],
          });
          this.searchErrorsChange.emit([]);

          this.loadRequest.set(null);
          if (this.initialLookupPending) {
            this.initialLookupPending = false;
            if (
              !this.rows.some(
                (row) =>
                  typeof row.standardApplicantId === 'number' &&
                  row.standardApplicantId === this.selectedId(),
              )
            ) {
              // Only the initial successful lookup can trigger this fallback.
              this.selectedIds = new Set();
              this.clearSearch();
            }
          }
        },
        onError: () => {
          this.initialLookupPending = false;
          this.rows = [];
          const errors: ErrorItem[] = [
            {
              id: 'standard-applicant-code',
              text: 'Unable to load standard applicants. Please try again.',
              href: '#standard-applicant-code',
            },
          ];
          this.saSignalState.patch({
            totalPages: 0,
            loading: false,
            searchErrors: errors,
          });
          this.searchErrorsChange.emit(errors);
          this.loadRequest.set(null);
        },
      },
      this.envInjector,
    );
  }

  private buildErrorSummary(): ErrorItem[] {
    return buildFormErrorSummary(this.form, this.errorMap, {
      hrefs: {
        code: '#standard-applicant-code',
        name: '#standard-applicant-name',
      },
    });
  }

  private getTrimmedFilters(): StandardApplicantFilters {
    const code = this.form.controls.code.value.trim();
    const name = this.form.controls.name.value.trim();

    return {
      code: code || undefined,
      name: name || undefined,
    };
  }

  private loadPage(
    page: number,
    filters: StandardApplicantFilters = this.appliedFilters,
  ): void {
    if (this.saState().loading) {
      return;
    }

    this.saSignalState.patch({ loading: true, pageIndex: page });

    const pageSize = this.saState().pageSize;
    const sort = this.saState().sortField;
    const apiSortKey = toStandardApplicantSortKey(sort.key);

    this.loadRequest.set({
      code: filters.code,
      name: filters.name,
      pageNumber: page,
      pageSize,
      sort: [`${apiSortKey},${sort.direction}`],
    });
  }

  private syncSelectedIds(): void {
    const id = this.selectedId();
    this.selectedIds =
      id === null || !this.selectedCode() ? new Set() : new Set([String(id)]);
    this.selectedRowCode = null;
  }
}
