import {
  ChangeDetectionStrategy,
  Component,
  forwardRef,
  input,
} from '@angular/core';
import {
  ControlValueAccessor,
  FormsModule,
  NG_VALUE_ACCESSOR,
} from '@angular/forms';

export interface ReportOption {
  id: string;
  label: string;
  hint?: string;
  disabled?: boolean;
}

@Component({
  selector: 'app-report-selector',
  standalone: true,
  imports: [FormsModule],
  providers: [
    {
      provide: NG_VALUE_ACCESSOR,
      useExisting: forwardRef(() => ReportSelectorComponent),
      multi: true,
    },
  ],
  changeDetection: ChangeDetectionStrategy.Eager,
  templateUrl: './report-selector.component.html',
})
export class ReportSelectorComponent implements ControlValueAccessor {
  /** Required: the list of radio options */
  readonly options = input.required<ReportOption[]>();
  /** Heading text in the legend */
  legend = input('Select an option');
  /** Radio group name (use a unique name per instance) */
  name = input('report');
  /** Prefix for the input/label IDs (unique per instance is ideal) */
  idPrefix = input('report');
  /** Optional aria-describedby id */
  ariaDescribedBy = input<string | null>(null);
  selectorDisabled = input(false);

  formDisabled = false;
  value: string | null = null;

  onChange: (v: string | null) => void = () => {};
  onTouched: () => void = () => {};

  writeValue(v: string | null): void {
    this.value = v;
  }

  registerOnChange(fn: (v: string | null) => void): void {
    this.onChange = fn;
  }

  registerOnTouched(fn: () => void): void {
    this.onTouched = fn;
  }

  setDisabledState(isDisabled: boolean): void {
    this.formDisabled = isDisabled;
  }

  onModelChange(v: string): void {
    this.value = v;
    this.onChange(v);
  }
}

<div class="govuk-form-group">
  <fieldset class="govuk-fieldset" [attr.aria-describedby]="ariaDescribedBy()">
    <legend class="govuk-fieldset__legend govuk-fieldset__legend--l">
      <h1 class="govuk-fieldset__heading">{{ legend() }}</h1>
    </legend>

    <div class="govuk-radios" data-module="govuk-radios">
      @for (r of options(); track r.id) {
        <div class="govuk-radios__item">
          <input
            class="govuk-radios__input"
            [id]="idPrefix() + '-' + r.id"
            [name]="name()"
            type="radio"
            [value]="r.id"
            [disabled]="formDisabled || selectorDisabled() || !!r.disabled"
            [attr.disabled]="
              formDisabled || selectorDisabled() || !!r.disabled ? true : null
            "
            [(ngModel)]="value"
            [ngModelOptions]="{ standalone: true }"
            (ngModelChange)="onModelChange($event)"
            (blur)="onTouched()"
          />
          <label
            class="govuk-label govuk-radios__label"
            [for]="idPrefix() + '-' + r.id"
          >
            {{ r.label }}
          </label>
          @if (r.hint) {
            <div
              class="govuk-hint govuk-radios__hint"
              [id]="idPrefix() + '-' + r.id + '-hint'"
            >
              {{ r.hint }}
            </div>
          }
        </div>
      } @empty {
        <p class="govuk-hint">No options.</p>
      }
    </div>
  </fieldset>
</div>
